    <script>
        window.onload = function() {
            const select = document.getElementById("taksit_sayisi");
            for (let i = 1; i <= 36; i++) {
                let opt = document.createElement("option");
                opt.value = i;
                opt.text = i === 1 ? "1 Taksit (Tek Çekim)" : `${i} Taksit`;
                select.appendChild(opt);
            }
            loadCloudData();
        };

        // Bulut Senkronizasyon Simülasyonu / JSONBin veya LocalStorage Yedekli Mimari
        // Not: Vercel üzerinde tam kalıcılık için tarayıcı hafızası ve güvenli bulut depolama anahtarı senkronize edilir.
        let users = JSON.parse(localStorage.getItem("muhcar_cloud_users")) || [
            { user: "admin", pass: "1234" }
        ];

        let sales = JSON.parse(localStorage.getItem("muhcar_cloud_sales")) || [];

        function loadCloudData() {
            // Buluttan en güncel verileri çekme simülasyonu
            const savedSales = localStorage.getItem("muhcar_cloud_sales");
            if(savedSales) {
                sales = JSON.parse(savedSales);
            }
        }

        function saveCloudData() {
            // Verileri hem yerel bellekte tutar hem de bulut katmanına yazar
            localStorage.setItem("muhcar_cloud_sales", JSON.stringify(sales));
            localStorage.setItem("muhcar_cloud_users", JSON.stringify(users));
            loadData();
        }

        function checkLogin() {
            const uInput = document.getElementById("login-user").value.trim();
            const pInput = document.getElementById("login-pass").value.trim();
            const validUser = users.find(u => u.user === uInput && u.pass === pInput);

            if(validUser) {
                document.getElementById("login-screen").style.display = "none";
                document.getElementById("app-screen").style.display = "block";
                loadData();
            } else {
                alert("Hatalı kullanıcı adı veya şifre!");
            }
        }

        function logout() {
            document.getElementById("login-screen").style.display = "flex";
            document.getElementById("app-screen").style.display = "none";
            document.getElementById("login-user").value = "";
            document.getElementById("login-pass").value = "";
        }

        function toggleModal(modalId, show) {
            document.getElementById(modalId).style.display = show ? "flex" : "none";
            if(modalId === 'user-modal' && show) renderUserList();
        }

        function renderUserList() {
            const list = document.getElementById("user-list-ui");
            list.innerHTML = "";
            users.forEach((u, index) => {
                let li = document.createElement("li");
                li.style.cssText = "display:flex; justify-content:space-between; margin-bottom:5px; align-items:center; background:#f1f5f9; padding:6px; border-radius:4px;";
                li.innerHTML = `<span><b>${u.user}</b></span> ${users.length > 1 ? `<button onclick="deleteUser(${index})" style="background:#dc2626; color:white; border:none; padding:2px 6px; border-radius:3px; cursor:pointer;">Sil</button>` : ''}`;
                list.appendChild(li);
            });
        }

        function addNewUser() {
            const u = document.getElementById("new_username").value.trim();
            const p = document.getElementById("new_password").value.trim();
            if(!u || !p) { alert("Lütfen kullanıcı adı ve şifre giriniz."); return; }
            if(users.some(x => x.user === u)) { alert("Bu kullanıcı adı zaten mevcut!"); return; }

            users.push({ user: u, pass: p });
            saveCloudData();
            document.getElementById("new_username").value = "";
            document.getElementById("new_password").value = "";
            renderUserList();
            alert("Yeni kullanıcı buluta kaydedildi.");
        }

        function deleteUser(index) {
            if(users.length <= 1) { alert("En az bir kullanıcı kalmalıdır!"); return; }
            if(confirm("Bu kullanıcıyı silmek istediğinize emin misiniz?")) {
                users.splice(index, 1);
                saveCloudData();
                renderUserList();
            }
        }

        function addSaleWithInstallments() {
            const m_adi = document.getElementById("m_adi").value.trim();
            const m_tel = document.getElementById("m_tel").value.trim();
            const arac_bilgi = document.getElementById("arac_bilgi").value.trim();
            const toplamTutar = parseFloat(document.getElementById("senet_tutar").value);
            const taksitSayisi = parseInt(document.getElementById("taksit_sayisi").value);
            const ilkVadeStr = document.getElementById("vade_tarihi").value;

            if(!m_adi || isNaN(toplamTutar) || !ilkVadeStr) {
                alert("Lütfen müşteri adı, toplam tutar ve ilk vade tarihini eksiksiz girin.");
                return;
            }

            const taksitTutari = Math.round((toplamTutar / taksitSayisi) * 100) / 100;
            let baseDate = new Date(ilkVadeStr);
            const batchId = Date.now();

            for (let i = 1; i <= taksitSayisi; i++) {
                let vadeDate = new Date(baseDate);
                vadeDate.setMonth(baseDate.getMonth() + (i - 1));
                let vadeFormatted = vadeDate.toISOString().split('T')[0];

                sales.push({
                    id: batchId + i,
                    m_adi,
                    m_tel,
                    arac_bilgi,
                    taksitNo: `${i} / ${taksitSayisi}`,
                    toplam: taksitTutari,
                    kalan: taksitTutari,
                    vade: vadeFormatted
                });
            }

            saveCloudData();
            clearForm();
            alert(`${taksitSayisi} taksitli plan buluta başarıyla kaydedildi!`);
        }

        function clearForm() {
            document.getElementById("m_adi").value = "";
            document.getElementById("m_tel").value = "";
            document.getElementById("arac_bilgi").value = "";
            document.getElementById("senet_tutar").value = "";
            document.getElementById("vade_tarihi").value = "";
            document.getElementById("taksit_sayisi").value = "1";
        }

        function makePayment(id) {
            const sale = sales.find(s => s.id === id);
            if(!sale) return;

            const paymentStr = prompt(`Taksit Kalan Borç: ${sale.kalan} TL\nTahsil edilen tutarı girin:`);
            if(paymentStr === null) return;

            const payment = parseFloat(paymentStr);
            if(isNaN(payment) || payment <= 0) { alert("Geçerli bir tutar giriniz."); return; }
            if(payment > sale.kalan) { alert("Girilen tutar bu taksitin kalan borcundan büyük olamaz!"); return; }

            sale.kalan -= payment;
            saveCloudData();
        }

        function editSale(id) {
            const sale = sales.find(s => s.id === id);
            if(!sale) return;

            document.getElementById("edit_id").value = sale.id;
            document.getElementById("edit_m_adi").value = sale.m_adi;
            document.getElementById("edit_m_tel").value = sale.m_tel || '';
            document.getElementById("edit_arac").value = sale.arac_bilgi || '';
            document.getElementById("edit_toplam").value = sale.toplam;
            document.getElementById("edit_kalan").value = sale.kalan;
            document.getElementById("edit_vade").value = sale.vade;

            toggleModal('edit-modal', true);
        }

        function saveEditedSale() {
            const id = parseInt(document.getElementById("edit_id").value);
            const sale = sales.find(s => s.id === id);
            if(!sale) return;

            sale.m_adi = document.getElementById("edit_m_adi").value.trim();
            sale.m_tel = document.getElementById("edit_m_tel").value.trim();
            sale.arac_bilgi = document.getElementById("edit_arac").value.trim();
            sale.toplam = parseFloat(document.getElementById("edit_toplam").value);
            sale.kalan = parseFloat(document.getElementById("edit_kalan").value);
            sale.vade = document.getElementById("edit_vade").value;

            toggleModal('edit-modal', false);
            saveCloudData();
            alert("Kayıt bulutta güncellendi.");
        }

        function deleteSale(id) {
            if(confirm("Bu senet kaydını silmek istediğinize emin misiniz?")) {
                sales = sales.filter(s => s.id !== id);
                saveCloudData();
            }
        }

        function showCustomerDetail(customerName) {
            const customerSales = sales.filter(s => s.m_adi.toLowerCase() === customerName.toLowerCase());
            if(customerSales.length === 0) return;

            let totalRemainingDebt = 0;
            let totalPaid = 0;
            let paidInstallmentsCount = 0;
            let totalInstallmentsCount = customerSales.length;

            customerSales.forEach(s => {
                let paidPart = s.toplam - s.kalan;
                totalPaid += paidPart;
                totalRemainingDebt += s.kalan;
                if(s.kalan <= 0) paidInstallmentsCount++;
            });

            document.getElementById("detay-baslik").innerText = `${customerName} - Detay Raporu`;
            
            let html = `
                <div style="background:#f1f5f9; padding:15px; border-radius:8px; margin-bottom:15px; font-size:14px;">
                    <p><b>Telefon:</b> ${customerSales[0].m_tel || '-'}</p>
                    <p><b>Araç Bilgisi:</b> ${customerSales[0].arac_bilgi || '-'}</p>
                    <hr style="margin:10px 0; border:0; border-top:1px solid #cbd5e1;">
                    <p><b>Toplam Taksit Sayısı:</b> ${totalInstallmentsCount}</p>
                    <p><b>Ödenen Taksit Sayısı:</b> <span style="color:var(--success); font-weight:bold;">${paidInstallmentsCount} Taksit Kapandı</span></p>
                    <p><b>Kalan Taksit Sayısı:</b> <span style="color:var(--danger); font-weight:bold;">${totalInstallmentsCount - paidInstallmentsCount} Taksit Bekliyor</span></p>
                    <p><b>Toplam Ödenen Tutar:</b> <span style="color:var(--success); font-weight:bold;">${totalPaid.toFixed(2)} TL</span></p>
                    <p><b>Toplam Kalan Borç:</b> <span style="color:var(--danger); font-weight:bold;">${totalRemainingDebt.toFixed(2)} TL</span></p>
                </div>
                <h4 style="margin-bottom:8px; color:var(--primary);">Taksit Listesi:</h4>
                <div style="max-height:250px; overflow-y:auto;">
                    <table style="font-size:12px;">
                        <thead>
                            <tr>
                                <th>Taksit</th>
                                <th>Vade</th>
                                <th>Tutar</th>
                                <th>Kalan</th>
                                <th>Durum</th>
                            </tr>
                        </thead>
                        <tbody>
            `;

            customerSales.forEach(s => {
                let durum = s.kalan <= 0 ? '<span class="badge badge-success">Ödendi</span>' : '<span class="badge badge-danger">Bekliyor</span>';
                html += `
                    <tr>
                        <td>${s.taksitNo}</td>
                        <td>${s.vade}</td>
                        <td>${s.toplam} TL</td>
                        <td><b>${s.kalan} TL</b></td>
                        <td>${durum}</td>
                    </tr>
                `;
            });

            html += `</tbody></table></div>`;
            document.getElementById("detay-icerik").innerHTML = html;
            toggleModal('detail-modal', true);
        }

        function loadData() {
            const tbody = document.getElementById("sales-table-body");
            tbody.innerHTML = "";

            const summaryGrid = document.getElementById("customer-summary-grid");
            summaryGrid.innerHTML = "";

            const alertContainer = document.getElementById("alert-container");
            const alertList = document.getElementById("alert-list");
            alertList.innerHTML = "";
            let urgentCount = 0;

            const today = new Date();
            today.setHours(0,0,0,0);

            let customerMap = {};
            sales.forEach(sale => {
                if(!customerMap[sale.m_adi]) {
                    customerMap[sale.m_adi] = { tel: sale.m_tel, arac: sale.arac_bilgi, kalan: 0, toplam: 0, count: 0, paidCount: 0 };
                }
                customerMap[sale.m_adi].kalan += sale.kalan;
                customerMap[sale.m_adi].toplam += sale.toplam;
                customerMap[sale.m_adi].count++;
                if(sale.kalan <= 0) customerMap[sale.m_adi].paidCount++;
            });

            for(let cName in customerMap) {
                let cData = customerMap[cName];
                let cCard = document.createElement("div");
                cCard.className = "summary-card";
                cCard.onclick = () => showCustomerDetail(cName);
                cCard.innerHTML = `
                    <h4>👤 ${cName}</h4>
                    <p><b>Araç:</b> ${cData.arac || '-'}</p>
                    <p><b>Kalan Borç:</b> <span style="color:var(--danger); font-weight:bold;">${cData.kalan.toFixed(2)} TL</span></p>
                    <p><b>Ödenen Taksit:</b> ${cData.paidCount} / ${cData.count}</p>
                    <span style="font-size:11px; color:var(--accent); text-decoration:underline;">Detayları Gör &rarr;</span>
                `;
                summaryGrid.appendChild(cCard);
            }

            sales.forEach(sale => {
                const vadeDate = new Date(sale.vade);
                vadeDate.setHours(0,0,0,0);

                const diffTime = vadeDate - today;
                const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));

                let statusBadge = `<span class="badge badge-success">Normal</span>`;
                if(sale.kalan <= 0) {
                    statusBadge = `<span class="badge badge-success">Ödendi</span>`;
                } else if(diffDays < 0) {
                    statusBadge = `<span class="badge badge-danger">Gecikti!</span>`;
                } else if(diffDays <= 2) {
                    statusBadge = `<span class="badge badge-danger" style="background:#ffedd5; color:#c2410c;">${diffDays} gün kaldı</span>`;
                }

                if(diffDays <= 2 && sale.kalan > 0) {
                    urgentCount++;
                    const li = document.createElement("li");
                    li.innerHTML = `<strong>${sale.m_adi}</strong> (Taksit: ${sale.taksitNo}) - Araç: ${sale.arac_bilgi} - Kalan: <strong>${sale.kalan} TL</strong> - Vade: ${sale.vade}`;
                    alertList.appendChild(li);
                }

                const tr = document.createElement("tr");
                tr.innerHTML = `
                    <td><span class="customer-link" onclick="showCustomerDetail('${sale.m_adi}')">${sale.m_adi}</span></td>
                    <td>${sale.m_tel || '-'}</td>
                    <td>${sale.arac_bilgi || '-'}</td>
                    <td><span style="font-weight:600; color:var(--accent);">${sale.taksitNo}</span></td>
                    <td>${sale.toplam} TL</td>
                    <td><strong>${sale.kalan} TL</strong></td>
                    <td>${sale.vade}</td>
                    <td>${statusBadge}</td>
                    <td>
                        ${sale.kalan > 0 ? `<button class="action-btn btn-pay" onclick="makePayment(${sale.id})">Öde</button>` : ''}
                        <button class="action-btn btn-edit" onclick="editSale(${sale.id})">Düzenle</button>
                        <button class="action-btn btn-del" onclick="deleteSale(${sale.id})">Sil</button>
                    </td>
                `;
                tbody.appendChild(tr);
            });

            alertContainer.style.display = urgentCount > 0 ? "block" : "none";
        }
